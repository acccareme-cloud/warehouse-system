AuthContext.jsx
import { createContext, useState, useContext, useEffect } from 'react';

const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);

  // Load user from localStorage on mount
  useEffect(() => {
    const storedUser = localStorage.getItem('user');
    const storedToken = localStorage.getItem('token');
    if (storedUser && storedToken) {
      try {
        setUser(JSON.parse(storedUser));
      } catch (e) {
        console.error('Error parsing user:', e);
      }
    }
  }, []);

  const login = (userData) => {
    setUser(userData);
    localStorage.setItem('token', userData.token);
    localStorage.setItem('user', JSON.stringify(userData));
  };

  const logout = () => {
    setUser(null);
    localStorage.removeItem('token');
    localStorage.removeItem('user');
  };

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);

import { useState } from 'react';
import api from '../services/api';
import { useAuth } from '../context/AuthContext';

function ChangePassword() {
  const { user } = useAuth();
  const [formData, setFormData] = useState({
    current_password: '',
    new_password: '',
    confirm_password: ''
  });
  const [message, setMessage] = useState('');
  const [error, setError] = useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage('');
    setError('');

    if (formData.new_password !== formData.confirm_password) {
      setError('كلمة السر الجديدة غير متطابقة');
      return;
    }

    if (formData.new_password.length < 6) {
      setError('كلمة السر يجب أن تكون 6 أحرف على الأقل');
      return;
    }

    try {
      await api.put('/auth/change-password', {
        current_password: formData.current_password,
        new_password: formData.new_password
      });
      setMessage('تم تغيير كلمة السر بنجاح');
      setFormData({ current_password: '', new_password: '', confirm_password: '' });
    } catch (err) {
      setError(err.response?.data?.message || 'حدث خطأ');
    }
  };

  return (
    <div style={styles.container}>
      <div style={styles.card}>
        <h1 style={styles.title}>🔐 تغيير كلمة السر</h1>
        <p style={styles.subtitle}>المستخدم: {user?.full_name || user?.username}</p>
        
        {message && <div style={styles.success}>{message}</div>}
        {error && <div style={styles.error}>{error}</div>}

        <form onSubmit={handleSubmit}>
          <div style={styles.inputGroup}>
            <label style={styles.label}>كلمة السر الحالية:</label>
            <input
              type="password"
              value={formData.current_password}
              onChange={(e) => setFormData({...formData, current_password: e.target.value})}
              required
              style={styles.input}
            />
          </div>
          
          <div style={styles.inputGroup}>
            <label style={styles.label}>كلمة السر الجديدة:</label>
            <input
              type="password"
              value={formData.new_password}
              onChange={(e) => setFormData({...formData, new_password: e.target.value})}
              required
              style={styles.input}
            />
          </div>
          
          <div style={styles.inputGroup}>
            <label style={styles.label}>تأكيد كلمة السر الجديدة:</label>
            <input
              type="password"
              value={formData.confirm_password}
              onChange={(e) => setFormData({...formData, confirm_password: e.target.value})}
              required
              style={styles.input}
            />
          </div>

          <button type="submit" style={styles.button}>
            💾 حفظ التغييرات
          </button>
        </form>

        <button 
          onClick={() => window.location.href = '/dashboard'}
          style={styles.backButton}
        >
          🔙 رجوع للوحة التحكم
        </button>
      </div>
    </div>
  );
}

                style={{ width: '100%', padding: '10px', border: '1px solid #ddd', borderRadius: '4px' }} 
              />
            </div>
          </div>

          <button 
            type="submit" 
            disabled={loading}
            style={{ 
              marginTop: '20px', 
              padding: '12px 40px', 
              backgroundColor: loading ? '#6c757d' : '#007bff', 
              color: 'white', 
              border: 'none', 
              borderRadius: '8px', 
              cursor: loading ? 'not-allowed' : 'pointer',
              fontSize: '16px', 
              fontWeight: 'bold'
            }}
          >
            {loading ? '⏳ جاري الإرسال...' : '📤 إرسال الطلب'}
          </button>
        </form>
      </div>

      {/* قائمة طلبات الشراء */}
      <h3>📋 جميع طلبات الشراء</h3>
      <div style={{ overflowX: 'auto' }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', marginTop: '10px', backgroundColor: 'white' }}>
          <thead>
            <tr style={{ backgroundColor: '#343a40', color: 'white' }}>
              <th style={thStyle}>رقم الطلب</th>
              <th style={thStyle}>التاريخ</th>
              <th style={thStyle}>القسم</th>
              <th style={thStyle}>الصنف</th>
              <th style={thStyle}>الكمية</th>
              <th style={thStyle}>طلب بواسطة</th>
              <th style={thStyle}>الحالة</th>
            </tr>
          </thead>
          <tbody>
            {requests.length === 0 ? (
              <tr>
                <td colSpan="7" style={{ textAlign: 'center', padding: '30px', color: '#6c757d' }}>
                  لا يوجد طلبات شراء
                </td>
              </tr>
            ) : (
              requests.map(r => (
                <tr key={r.id} style={{ backgroundColor: r.id % 2 === 0 ? '#f8f9fa' : 'white' }}>
                  <td style={tdStyle}><strong>{r.request_number}</strong></td>
                  <td style={tdStyle}>{new Date(r.request_date).toLocaleDateString('ar-EG')}</td>
                  <td style={tdStyle}>{r.department}</td>
                  <td style={tdStyle}>{r.item_code} - {r.item_name}</td>
                  <td style={tdStyle}>{r.quantity}</td>
                  <td style={tdStyle}>{r.requested_by_name}</td>
                  <td style={tdStyle}>
                    <span style={{ 
                      color: getStatusColor(r.status),
                      fontWeight: 'bold',
                      padding: '4px 12px',
                      borderRadius: '12px',
                      backgroundColor: getStatusColor(r.status) + '20'
                    }}>
                      {getStatusText(r.status)}
                    </span>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}

Purchases.jsx

import { useState, useEffect } from 'react';
import api from '../services/api';

function Purchases() {
  const [activeTab, setActiveTab] = useState('local');
  const [items, setItems] = useState([]);
  const [warehouses, setWarehouses] = useState([]);
  const [purchases, setPurchases] = useState([]);
  const [showForm, setShowForm] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [userRole, setUserRole] = useState('');

  const [taxControls, setTaxControls] = useState({
    has_vat: true,
    has_discount_tax: true
  });

  const [formData, setFormData] = useState({
    purchase_number: '',
    supplier: '',
    item_id: '',
    warehouse_id: '',
    quantity: 1,
    unit_price: 0,
    tax_discount_percent: 0,
    final_release_value: 0,
    import_tax: 0,
    shipping_cost: 0,
    customs_duty: 0,
    customs_vat: 0,
    clearance_fees: 0,
    other_fees: 0,
    transfer_commission_type: 'percentage',
    transfer_commission_value: 0
  });

  const [extraExpenses, setExtraExpenses] = useState([
    { id: 1, description: 'تكلفة الشحنه بالدولار', amount_usd: 0, exchange_rate: 0, amount_egp: 0 },
    { id: 2, description: 'تكلفة الشحنه بالجنيه', amount_egp: 0 },
    { id: 3, description: 'مصاريف الشحنه', amount_egp: 0 },
    { id: 4, description: 'اجمالى مصاريف الشحنه', amount_egp: 0 },
    { id: 5, description: 'تكلفه الوحده بالدولار', amount_usd: 0, exchange_rate: 0, amount_egp: 0 },
    { id: 6, description: 'تكلفه الوحده بالجنيه', amount_egp: 0 },
    { id: 7, description: 'سعر معامل التحويل', amount_egp: 0 },
    { id: 8, description: 'مصاريف بحرى', amount_egp: 0 },
    { id: 9, description: 'غازات', amount_egp: 0 },
    { id: 10, description: 'مصاريف ادارية', amount_egp: 0 },
    { id: 11, description: 'تخليص', amount_egp: 0 },
    { id: 12, description: 'نقل', amount_egp: 0 },
    { id: 13, description: 'اكرامية استعجال', amount_egp: 0 },
    { id: 14, description: 'موازين', amount_egp: 0 },
    { id: 15, description: 'بوابه خروج', amount_egp: 0 },
    { id: 16, description: 'عروض', amount_egp: 0 },
    { id: 17, description: 'تجديد رقم التعامل', amount_egp: 0 },
    { id: 18, description: 'موافقه سياسات', amount_egp: 0 },
    { id: 19, description: 'موافقه استراديه', amount_egp: 0 },
    { id: 20, description: 'مصاريف بنكيه', amount_egp: 0 },
  ]);

  const [taxCalculations, setTaxCalculations] = useState({
    final_release_value: 0,
    import_tax: 0,
    total_with_tax: 0,
    tax_14_percent: 0,
    commercial_profit_tax: 0,
    transfer_commission_amount: 0
  });

  const [calculations, setCalculations] = useState({
    subtotal: 0,
    tax14: 0,
    taxDiscount: 0,
    netAmount: 0,
    totalShipment: 0,
    landedCost: 0
  });

  const [message, setMessage] = useState('');
  const [selectedFiles, setSelectedFiles] = useState([]);

  useEffect(() => {
    const user = JSON.parse(localStorage.getItem('user') || '{}');
    setUserRole(user.role || '');
    fetchItems();
    fetchWarehouses();
    fetchPurchases();
  }, [activeTab]);

  useEffect(() => {
    calculateTaxes();
  }, [formData.final_release_value, formData.import_tax, formData.transfer_commission_type, formData.transfer_commission_value]);

  useEffect(() => {
    calculateTotals();
  }, [formData.quantity, formData.unit_price, formData.tax_discount_percent, 
      formData.shipping_cost, formData.customs_duty, formData.customs_vat, 
      formData.clearance_fees, formData.other_fees, taxCalculations, extraExpenses, taxControls]);

  const fetchItems = async () => {
    try {
      const response = await api.get('/items');
      setItems(response.data);
    } catch (err) {
      console.error('خطأ في تحميل الاصناف');
    }
  };

  const fetchWarehouses = async () => {
    try {
      const response = await api.get('/warehouses');
      setWarehouses(response.data);
    } catch (err) {
      console.error('خطأ في تحميل المخازن');
    }
  };

  const fetchPurchases = async () => {
    try {
      const response = await api.get('/purchases');
      setPurchases(response.data.filter(p => p.purchase_type === activeTab));
    } catch (err) {
      console.error('خطأ في تحميل المشتريات');
    }
  };

  const fetchNextNumber = async () => {
    try {
      const type = activeTab === 'import' ? 'import' : 'local';
      const response = await api.get(`/purchases/next-number?type=${type}`);
      setFormData(prev => ({...prev, purchase_number: response.data.nextNumber}));
    } catch (err) {
      console.error('خطأ في توليد الرقم');
    }
  };

  const handleTabChange = (tab) => {
    setActiveTab(tab);
    setShowForm(false);
    setEditingId(null);
    fetchPurchases();
  };

  const handleShowForm = () => {
    setEditingId(null);
    setShowForm(true);
    fetchNextNumber();
    resetForm();
  };

  const handleShowList = () => {
    setShowForm(false);
    setEditingId(null);
    fetchPurchases();
  };

  const calculateTaxes = () => {
    if (activeTab !== 'import') return;

    const finalRelease = parseFloat(formData.final_release_value) || 0;
    const importTax = parseFloat(formData.import_tax) || 0;
    const totalWithTax = finalRelease + importTax;
    const tax14 = totalWithTax * 0.14;
    const commercialProfitTax = finalRelease * 0.01;

    let transferCommAmount = 0;
    if (formData.transfer_commission_type === 'percentage') {
      transferCommAmount = finalRelease * (parseFloat(formData.transfer_commission_value) || 0) / 100;
    } else {
      transferCommAmount = parseFloat(formData.transfer_commission_value) || 0;
    }

    setTaxCalculations({
      final_release_value: finalRelease,
      import_tax: importTax,
      total_with_tax: totalWithTax,
      tax_14_percent: tax14,
      commercial_profit_tax: commercialProfitTax,
      transfer_commission_amount: transferCommAmount
    });
  };

  const calculateExtraExpenses = () => {
    let total = 0;
    extraExpenses.forEach(item => {
      if (item.amount_egp) total += parseFloat(item.amount_egp) || 0;
      if (item.amount_usd && item.exchange_rate) {
        total += parseFloat(item.amount_usd) * parseFloat(item.exchange_rate);
      }
    });
    return total;
  };

  const handleExtraExpenseChange = (id, field, value) => {
    setExtraExpenses(prev => prev.map(item => {
      if (item.id === id) {
        const updated = { ...item, [field]: value };
        if (field === 'amount_usd' || field === 'exchange_rate') {
          if (updated.amount_usd && updated.exchange_rate) {
            updated.amount_egp = (parseFloat(updated.amount_usd) * parseFloat(updated.exchange_rate)).toFixed(2);
          }
        }
        return updated;
      }
      return item;
    }));
  };

  const calculateTotals = () => {
    const qty = parseFloat(formData.quantity) || 0;
    const price = parseFloat(formData.unit_price) || 0;
    const subtotal = qty * price;

    const tax14 = taxControls.has_vat ? subtotal * 0.14 : 0;

    const taxDiscountRate = taxControls.has_discount_tax ? (parseFloat(formData.tax_discount_percent) || 0) : 0;
    const taxDiscount = subtotal * (taxDiscountRate / 100);
    const netAmount = subtotal + tax14 - taxDiscount;

    const shipCost = parseFloat(formData.shipping_cost) || 0;
    const customsDuty = parseFloat(formData.customs_duty) || 0;
    const customsVat = parseFloat(formData.customs_vat) || 0;
    const clearanceFees = parseFloat(formData.clearance_fees) || 0;
    const otherFees = parseFloat(formData.other_fees) || 0;

    const extraTotal = calculateExtraExpenses();
    const totalTaxes = taxCalculations.import_tax + taxCalculations.tax_14_percent + 
                       taxCalculations.commercial_profit_tax + taxCalculations.transfer_commission_amount;

    const totalShipment = shipCost + customsDuty + customsVat + clearanceFees + otherFees + extraTotal + totalTaxes;
    const landedCost = qty > 0 ? price + (totalShipment / qty) : 0;

    setCalculations({
      subtotal: subtotal.toFixed(2),
      tax14: tax14.toFixed(2),
      taxDiscount: taxDiscount.toFixed(2),
      netAmount: netAmount.toFixed(2),
      totalShipment: totalShipment.toFixed(2),
      landedCost: landedCost.toFixed(2)
    });
  };

  const handleFileChange = (e) => {
    setSelectedFiles(Array.from(e.target.files));
  };

  const uploadAttachments = async (purchaseId, type) => {
    for (const file of selectedFiles) {
      const formDataFile = new FormData();
      formDataFile.append('file', file);
      formDataFile.append('reference_type', type);
      formDataFile.append('reference_id', purchaseId);
      formDataFile.append('description', file.name);

      try {
        await api.post('/attachments/upload', formDataFile, {
          headers: { 'Content-Type': 'multipart/form-data' }
        });
      } catch (err) {
        console.error('Error uploading file:', err);
      }
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      const dataToSend = {
        ...formData,
        purchase_type: activeTab,
        has_vat: taxControls.has_vat,
        has_discount_tax: taxControls.has_discount_tax,
        ...taxCalculations,
        insurance_cost: extraExpenses[0].amount_egp || 0,
        loading_cost: extraExpenses[2].amount_egp || 0,
        unloading_cost: extraExpenses[3].amount_egp || 0,
        delivery_cost: extraExpenses[1].amount_egp || 0,
        transport_cost: extraExpenses[11].amount_egp || 0,
        maritime_fees: extraExpenses[7].amount_egp || 0,
        gas_cost: extraExpenses[8].amount_egp || 0,
        admin_fees: extraExpenses[9].amount_egp || 0,
        port_fees: extraExpenses[13].amount_egp || 0,
        exit_gate_fees: extraExpenses[14].amount_egp || 0,
        offers_cost: extraExpenses[15].amount_egp || 0,
        renewal_fees: extraExpenses[16].amount_egp || 0,
        political_approvals: extraExpenses[17].amount_egp || 0,
        strategic_approvals: extraExpenses[18].amount_egp || 0,
        bank_fees: extraExpenses[19].amount_egp || 0
      };

      if (editingId) {
        // تعديل فاتورة موجودة
        await api.put(`/purchases/${editingId}`, dataToSend);
        setMessage('تم تحديث الفاتورة بنجاح');
      } else {
        // فاتورة جديدة
        const response = await api.post('/purchases', dataToSend);
        if (selectedFiles.length > 0) {
          await uploadAttachments(response.data.id, 'purchase');
        }
        setMessage('تم إنشاء الفاتورة بنجاح');
      }

      setShowForm(false);
      setEditingId(null);
      setSelectedFiles([]);
      resetForm();
      fetchPurchases();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const handleApprove = async (id, status) => {
    try {
      await api.put(`/purchases/${id}/approve`, { status });
      setMessage(status === 'approved' ? 'تم الاعتماد بنجاح' : 'تم الرفض بنجاح');
      fetchPurchases();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const handleSendToQuality = async (id) => {
    try {
      await api.put(`/purchases/${id}/quality`);
      setMessage('تم إرسال الفاتورة للجودة');
      fetchPurchases();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const handleQualityApprove = async (id) => {
    try {
      await api.put(`/purchases/${id}/quality-approve`);
      setMessage('تم اعتماد الجودة');
      fetchPurchases();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const handleDelete = async (id) => {
    if (!window.confirm('هل أنت متأكد من حذف هذه الفاتورة؟')) return;
    try {
      await api.delete(`/purchases/${id}`);
      setMessage('تم حذف الفاتورة');
      fetchPurchases();
    } catch (err) {
      setMessage('خطأ في حذف الفاتورة');
    }
  };

  const handleEdit = (purchase) => {
    setEditingId(purchase.id);
    setFormData({
      purchase_number: purchase.purchase_number,
      supplier: purchase.supplier || '',
      item_id: purchase.item_id,
      warehouse_id: purchase.warehouse_id,
      quantity: purchase.quantity,
      unit_price: purchase.unit_price,
      tax_discount_percent: purchase.tax_discount_percent || 0,
      final_release_value: purchase.final_release_value || 0,
      import_tax: purchase.import_tax || 0,
      shipping_cost: purchase.shipping_cost || 0,
      customs_duty: purchase.customs_duty || 0,
      customs_vat: purchase.customs_vat || 0,
      clearance_fees: purchase.clearance_fees || 0,
      other_fees: purchase.other_fees || 0,
      transfer_commission_type: purchase.transfer_commission_type || 'percentage',
      transfer_commission_value: purchase.transfer_commission_value || 0
    });

    setTaxControls({
      has_vat: purchase.has_vat !== false,
      has_discount_tax: purchase.has_discount_tax !== false
    });

    setShowForm(true);
  };

  const resetForm = () => {
    setEditingId(null);
    setFormData({
      purchase_number: '',
      supplier: '',
      item_id: '',
      warehouse_id: '',
      quantity: 1,
      unit_price: 0,
      tax_discount_percent: 0,
      final_release_value: 0,
      import_tax: 0,
      shipping_cost: 0,
      customs_duty: 0,
      customs_vat: 0,
      clearance_fees: 0,
      other_fees: 0,
      transfer_commission_type: 'percentage',
      transfer_commission_value: 0
    });
    setTaxControls({
      has_vat: true,
      has_discount_tax: true
    });
  };

  const getStatusText = (status) => {
    const statuses = {
      'draft': '✏️ مسودة',
      'pending': '⏳ بانتظار الاعتماد',
      'approved': '✓ معتمد',
      'pending_quality': '🔍 بانتظار الجودة',
      'quality_passed': '✓ جودة معتمدة',
      'rejected': '✕ مرفوض'
    };
    return statuses[status] || status;
  };

  const getStatusColor = (status) => {
    const colors = {
      'draft': '#6c757d',
      'pending': '#ffc107',
      'approved': '#17a2b8',
      'pending_quality': '#fd7e14',
      'quality_passed': '#28a745',
      'rejected': '#dc3545'
    };
    return colors[status] || '#6c757d';
  };

  const thStyle = { padding: '12px', border: '1px solid #ddd' };
  const tdStyle = { padding: '10px', border: '1px solid #ddd' };

  return (
    <div style={{ padding: '20px', maxWidth: '1400px', margin: '0 auto', direction: 'rtl' }}>
      <h1>🧾 فواتير المشتريات</h1>

      <button 
        onClick={() => window.location.href = '/dashboard'}
        style={{ padding: '10px 20px', backgroundColor: '#6c757d', color: 'white', border: 'none', borderRadius: '4px', marginBottom: '20px', cursor: 'pointer' }}
      >
        ← رجوع للوحة التحكم
      </button>

      {message && <p style={{ padding: '10px', backgroundColor: message.includes('نجاح') ? '#d4edda' : '#f8d7da', borderRadius: '4px', fontWeight: 'bold' }}>{message}</p>}

      {/* Tabs */}
      <div style={{ marginBottom: '20px', display: 'flex', gap: '10px', flexWrap: 'wrap' }}>
        <button
          onClick={() => handleTabChange('local')}
          style={{ 
            padding: '12px 30px', 
            backgroundColor: activeTab === 'local' ? '#0d9488' : '#e2e8f0', 
            color: activeTab === 'local' ? 'white' : '#333',
            border: 'none', 
            borderRadius: '8px', 
            cursor: 'pointer',
            fontSize: '16px',
            fontWeight: 'bold'
          }}
        >
          🏠 فاتورة محلية
        </button>
        <button
          onClick={() => handleTabChange('import')}
          style={{ 
            padding: '12px 30px', 
            backgroundColor: activeTab === 'import' ? '#92400e' : '#e2e8f0', 
            color: activeTab === 'import' ? 'white' : '#333',
            border: 'none', 
            borderRadius: '8px', 
            cursor: 'pointer',
            fontSize: '16px',
            fontWeight: 'bold'
          }}
        >
          🚢 فاتورة استيراد
        </button>
      </div>

      {/* Buttons */}
      <div style={{ marginBottom: '20px', display: 'flex', gap: '10px', flexWrap: 'wrap' }}>
        <button
          onClick={handleShowList}
          style={{ 
            padding: '12px 25px', 
            backgroundColor: !showForm ? '#6c757d' : '#e2e8f0', 
            color: !showForm ? 'white' : '#333',
            border: 'none', 
            borderRadius: '8px', 
            cursor: 'pointer',
            fontSize: '16px',
            fontWeight: 'bold'
          }}
        >
          📋 عرض الفواتير
        </button>
        <button
          onClick={handleShowForm}
          style={{ 
            padding: '12px 25px', 
            backgroundColor: showForm ? '#28a745' : '#e2e8f0', 
            color: showForm ? 'white' : '#333',
            border: 'none', 
            borderRadius: '8px', 
            cursor: 'pointer',
            fontSize: '16px',
            fontWeight: 'bold'
          }}
        >
          ➕ فاتورة {activeTab === 'local' ? 'محلية' : 'استيراد'} جديدة
        </button>
      </div>

      {/* Form */}
      {showForm && (
        <form onSubmit={handleSubmit} style={{ backgroundColor: '#f8f9fa', padding: '20px', borderRadius: '8px', marginBottom: '20px', border: `3px solid ${activeTab === 'import' ? '#92400e' : '#0d9488'}` }}>
          <h3 style={{ color: activeTab === 'import' ? '#92400e' : '#0d9488' }}>
            {editingId ? '✏️ تعديل فاتورة' : `➕ فاتورة ${activeTab === 'local' ? 'محلية' : 'استيراد'} جديدة`}
          </h3>

          {/* تحكم الضرائب للمدير المالي */}
          {['finance', 'admin'].includes(userRole) && (
            <div style={{ 
              marginBottom: '20px', 
              padding: '15px', 
              backgroundColor: '#fff3cd', 
              borderRadius: '8px',
              border: '2px solid #ffc107'
            }}>
              <h4 style={{ marginBottom: '10px', color: '#856404' }}>⚙️ تحكم الضرائب (المدير المالي)</h4>
              <div style={{ display: 'flex', gap: '20px', flexWrap: 'wrap' }}>
                <label style={{ display: 'flex', alignItems: 'center', gap: '8px', cursor: 'pointer' }}>
                  <input 
                    type="checkbox" 
                    checked={taxControls.has_vat}
                    onChange={(e) => setTaxControls({...taxControls, has_vat: e.target.checked})}
                  />
                  <span>تفعيل ضريبة القيمة المضافة (14%)</span>
                </label>
                <label style={{ display: 'flex', alignItems: 'center', gap: '8px', cursor: 'pointer' }}>
                  <input 
                    type="checkbox" 
                    checked={taxControls.has_discount_tax}
                    onChange={(e) => setTaxControls({...taxControls, has_discount_tax: e.target.checked})}
                  />
                  <span>تفعيل ضريبة الخصم</span>
                </label>
              </div>
              {!taxControls.has_vat && <p style={{color: '#dc3545', marginTop: '5px'}}>⚠️ ضريبة القيمة المضافة معطلة</p>}
              {!taxControls.has_discount_tax && <p style={{color: '#dc3545', marginTop: '5px'}}>⚠️ ضريبة الخصم معطلة</p>}
            </div>
          )}

          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: '15px' }}>

import { useState, useEffect } from 'react';
import api from '../services/api';

function Quality() {
  const [pendingRequests, setPendingRequests] = useState([]);
  const [pendingReceipts, setPendingReceipts] = useState([]);
  const [pendingPurchases, setPendingPurchases] = useState([]); // ✅ جديد
  const [activeTab, setActiveTab] = useState('requests');
  const [formData, setFormData] = useState({
    reference_type: 'request',
    reference_id: '',
    result: 'passed',
    notes: ''
  });
  const [message, setMessage] = useState('');

  useEffect(() => {
    fetchPendingRequests();
    fetchPendingReceipts();
    fetchPendingPurchases(); // ✅ جديد
  }, []);

  const fetchPendingRequests = async () => {
    try {
      const response = await api.get('/requests/pending');
      setPendingRequests(response.data);
    } catch (err) {
      console.error('خطأ في تحميل الطلبات');
    }
  };

  const fetchPendingReceipts = async () => {
    try {
      const response = await api.get('/receipts/pending');
      setPendingReceipts(response.data);
    } catch (err) {
      console.error('خطأ في تحميل الاذونات');
    }
  };

  // ✅ جديد: جلب الفواتير بانتظار الجودة
  const fetchPendingPurchases = async () => {
    try {
      const response = await api.get('/purchases');
      const pending = response.data.filter(p => p.status === 'pending_quality');
      setPendingPurchases(pending);
    } catch (err) {
      console.error('خطأ في تحميل الفواتير');
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      await api.post('/quality', formData);
      setMessage('تم فحص الجودة بنجاح');
      setFormData({ reference_type: 'request', reference_id: '', result: 'passed', notes: '' });
      fetchPendingRequests();
      fetchPendingReceipts();
      fetchPendingPurchases(); // ✅ جديد
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const handleQualityApprove = async (id) => {
    try {
      await api.put(`/purchases/${id}/quality-approve`);
      setMessage('تم اعتماد الجودة بنجاح');
      fetchPendingPurchases();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }

  const handleQualityReject = async (id) => {
    try {
      await api.put(`/purchases/${id}/approve`, { status: 'rejected' });

      <button 
        onClick={() => window.location.href = '/dashboard'}
        style={{ padding: '10px 20px', backgroundColor: '#6c757d', color: 'white', border: 'none', borderRadius: '4px', marginBottom: '20px', cursor: 'pointer' }}
      >
        ← رجوع للوحة التحكم

        <button 
          onClick={() => setActiveTab('requests')}

          📋 طلبات الصرف ({pendingRequests.length})
        </button>
        <button 
          onClick={() => setActiveTab('receipts')}
          style={{ padding: '10px 20px', backgroundColor: activeTab === 'receipts' ? '#0891b2' : '#e2e8f0', color: activeTab === 'receipts' ? 'white' : '#333', border: 'none', borderRadius: '4px', cursor: 'pointer', fontWeight: 'bold' }}

    item_id: '',

    supply_order: '',
    receipt_date: new Date().toISOString().split('T')[0],
    has_serial: false
  });
  
  const [serials, setSerials] = useState([]);
  const [calculations, setCalculations] = useState({
    subtotal: 0,
    tax14: 0,
    taxDiscount: 0,
    total: 0
  });
  const [message, setMessage] = useState('');

  useEffect(() => {
    fetchItems();
    fetchWarehouses();
    fetchReceipts();
    fetchNextNumber();
  }, []);

  useEffect(() => {
    calculateTotals();
  }, [formData.quantity, formData.purchase_price, formData.tax_discount_percent]);

  useEffect(() => {
    if (formData.has_serial) {
      const qty = parseInt(formData.quantity) || 0;
      const newSerials = Array(qty).fill('').map((_, i) => serials[i] || '');
      setSerials(newSerials);
    } else {
      setSerials([]);
    }
  }, [formData.quantity, formData.has_serial]);

  const fetchNextNumber = async () => {
    try {
      const response = await api.get('/receipts/next-number');
      setFormData(prev => ({...prev, voucher_number: response.data.nextNumber}));
    } catch (err) {
      console.error('خطأ في توليد رقم الإذن');
    }

    const qty = parseFloat(formData.quantity) || 0;
    const price = parseFloat(formData.purchase_price) || 0;
    const subtotal = qty * price;
    const tax14 = subtotal * 0.14;
    const taxDiscountRate = parseFloat(formData.tax_discount_percent) || 0;

    const total = subtotal + tax14 - taxDiscount;

      subtotal: subtotal.toFixed(2),
      tax14: tax14.toFixed(2),
      taxDiscount: taxDiscount.toFixed(2),
      total: total.toFixed(2)
    });

    const newSerials = [...serials];

        return;

        supply_order: '',

        has_serial: false

import { useState, useEffect } from 'react';
import api from '../services/api';

function Requests() {
  const [items, setItems] = useState([]);
  const [warehouses, setWarehouses] = useState([]);
  const [requests, setRequests] = useState([]);
  const [availableSerials, setAvailableSerials] = useState([]);
  const [selectedSerials, setSelectedSerials] = useState([]);
  
  const [formData, setFormData] = useState({
    request_number: '',
    department: '',
    item_id: '',
    warehouse_id: '',
    quantity: 1,
    work_order: '',
    customer_name: '',
    salesperson: '',
    notes: ''
  });
  
  const [message, setMessage] = useState('');

  useEffect(() => {
    fetchItems();
    fetchWarehouses();
    fetchRequests();
    generateRequestNumber();
  }, []);

  // لما يختار الصنف والمخزن، نجيب السريالات المتوفرة
  useEffect(() => {
    if (formData.item_id && formData.warehouse_id) {
      fetchAvailableSerials();
    } else {
      setAvailableSerials([]);
      setSelectedSerials([]);
    }
  }, [formData.item_id, formData.warehouse_id, formData.quantity]);

  const generateRequestNumber = () => {
    const date = new Date().toISOString().slice(0, 10).replace(/-/g, '');
    const random = Math.floor(Math.random() * 1000).toString().padStart(3, '0');
    setFormData(prev => ({...prev, request_number: `REQ-${date}-${random}`}));
  };

  const fetchAvailableSerials = async () => {
    try {
      const response = await api.get(`/serials/available?item_id=${formData.item_id}&warehouse_id=${formData.warehouse_id}`);
      setAvailableSerials(response.data);
    } catch (err) {
      console.error('خطأ في تحميل السريالات:', err);
      setAvailableSerials([]);
    }
  };

  const handleSerialSelect = (serialId) => {
    setSelectedSerials(prev => {
      if (prev.includes(serialId)) {
        return prev.filter(id => id !== serialId);
      }
      if (prev.length >= parseInt(formData.quantity)) {
        setMessage('لا يمكن اختيار أكثر من ' + formData.quantity + ' سريال');
        setTimeout(() => setMessage(''), 3000);
        return prev;
      }
      return [...prev, serialId];
    });
  };

  const fetchItems = async () => {
    try {
      const response = await api.get('/items');
      setItems(response.data);
    } catch (err) {
      console.error('خطأ في تحميل الاصناف');
    }
  };

  const fetchWarehouses = async () => {
    try {
      const response = await api.get('/warehouses');
      setWarehouses(response.data);
    } catch (err) {
      console.error('خطأ في تحميل المخازن');
    }
  };

  const fetchRequests = async () => {
    try {
      const response = await api.get('/requests/my-requests');
      setRequests(response.data);
    } catch (err) {
      console.error('خطأ في تحميل الطلبات');
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    
    // التحقق من السريالات
    const selectedItem = items.find(i => i.id == formData.item_id);
    if (selectedItem?.has_serial && selectedSerials.length !== parseInt(formData.quantity)) {
      setMessage('خطأ: يجب اختيار ' + formData.quantity + ' سريال');
      return;
    }
    
    try {
      const dataToSend = {
        ...formData,
        selected_serials: selectedSerials
      };
      
      await api.post('/requests', dataToSend);
      setMessage('تم ارسال طلب الصرف بنجاح');
      
      setFormData({
        request_number: '',
        department: '',
        item_id: '',
        warehouse_id: '',
        quantity: 1,
        work_order: '',
        customer_name: '',
        salesperson: '',
        notes: ''
      });
      setSelectedSerials([]);
      setAvailableSerials([]);
      
      generateRequestNumber();
      fetchRequests();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));
    }
  };

  const thStyle = { padding: '12px', border: '1px solid #ddd' };
  const tdStyle = { padding: '10px', border: '1px solid #ddd' };

  // نجيب معلومات الصنف المختار
  const selectedItem = items.find(i => i.id == formData.item_id);

  return (
    <div style={{ padding: '20px', maxWidth: '1400px', margin: '0 auto' }}>
      <h1>طلب صرف من المخزن</h1>
      
      <button 
        onClick={() => window.location.href = '/dashboard'}
        style={{ padding: '10px 20px', backgroundColor: '#6c757d', color: 'white', border: 'none', borderRadius: '4px', marginBottom: '20px', cursor: 'pointer' }}
      >
        رجوع للوحة التحكم
      </button>

      {message && <p style={{ padding: '10px', backgroundColor: message.includes('نجاح') ? '#d4edda' : '#f8d7da', borderRadius: '4px' }}>{message}</p>}

      <form onSubmit={handleSubmit} style={{ backgroundColor: '#f8f9fa', padding: '20px', borderRadius: '8px', marginBottom: '20px' }}>
        <h3>طلب صرف جديد</h3>
        
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: '15px' }}>
          <div>
            <label>رقم الطلب (تلقائي):</label>
            <div style={{ display: 'flex', gap: '5px' }}>
              <input 
                type="text" 
                value={formData.request_number} 
                readOnly
                style={{ flex: 1, padding: '8px', backgroundColor: '#e2e8f0' }} 
              />
              <button 
                type="button" 
                onClick={generateRequestNumber} 
                style={{ padding: '8px 15px', backgroundColor: '#17a2b8', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer' }}
              >
                توليد
              </button>
            </div>
          </div>
          <div>
            <label>القسم:</label>
            <select value={formData.department} onChange={(e) => setFormData({...formData, department: e.target.value})} required style={{ width: '100%', padding: '8px' }}>
              <option value="">اختر القسم</option>
              <option value="المبيعات">المبيعات</option>
              <option value="الصيانة">الصيانة</option>
              <option value="الإنتاج">الإنتاج</option>
              <option value="المخازن">المخازن</option>
              <option value="الإدارة">الإدارة</option>
            </select>
          </div>
          <div>
            <label>الصنف:</label>
            <select value={formData.item_id} onChange={(e) => setFormData({...formData, item_id: e.target.value, quantity: 1, selected_serials: []})} required style={{ width: '100%', padding: '8px' }}>
              <option value="">اختر الصنف</option>
              {items.map(item => (
                <option key={item.id} value={item.id}>
                  {item.code} - {item.name} {item.has_serial ? '✓' : ''}
                </option>
              ))}
            </select>
          </div>
          <div>
            <label>المخزن:</label>
            <select value={formData.warehouse_id} onChange={(e) => setFormData({...formData, warehouse_id: e.target.value})} required style={{ width: '100%', padding: '8px' }}>

              value={formData.quantity} 
              onChange={(e) => {
                const qty = parseInt(e.target.value) || 1;
                setFormData({...formData, quantity: qty});
                setSelectedSerials([]); // إعادة تعيين السريالات المختارة
              }} 
              required 
              style={{ width: '100%', padding: '8px' }} 
            />
          </div>
          <div>
            <label>رقم أمر العمل (اختياري):</label>
            <input type="text" value={formData.work_order} onChange={(e) => setFormData({...formData, work_order: e.target.value})} style={{ width: '100%', padding: '8px' }} placeholder="مثال: WO-2024-001" />
          </div>
          
          <div>
            <label>اسم العميل:</label>
            <input 
              type="text" 
              value={formData.customer_name} 
              onChange={(e) => setFormData({...formData, customer_name: e.target.value})} 
              required 
              style={{ width: '100%', padding: '8px' }}

              value={formData.salesperson} 
              onChange={(e) => setFormData({...formData, salesperson: e.target.value})} 
              required 
              style={{ width: '100%', padding: '8px' }} 
              placeholder="اسم موظف المبيعات"

          </div>
          <div>
            <label>ملاحظات:</label>

            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: '10px' }}>

        </tbody>
      </table>
    </div>

}

function CostCenters() {

    center_code: '',

  });

  useEffect(() => {
    fetchCenters();
  }, []);

  const fetchCenters = async () => {
    try {
      const res = await api.get('/cost-centers');

    } catch (err) {
      console.error('خطأ في تحميل المراكز');
    }
  };

  const fetchNextCode = async () => {
    try {
      const res = await api.get(`/cost-centers/next-code?type=${formData.center_type}`);
      setFormData(prev => ({...prev, center_code: res.data.nextCode}));
    } catch (err) {
      console.error('خطأ في توليد الكود');
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      await api.post('/cost-centers', formData);
      setMessage('تم إنشاء مركز التكلفة بنجاح');
      setShowForm(false);
      fetchCenters();
    } catch (err) {
      setMessage('خطأ: ' + (err.response?.data?.message || 'حدث خطأ'));

      'admin': 'مصاريف إدارية',

      'operational': '#0d9488',
      'admin': '#2563eb',
      'production': '#dc3545'
    };
    return colors[type] || '#6c757d';
  };

  const thStyle = { padding: '12px', border: '1px solid #ddd' };
  const tdStyle = { padding: '10px', border: '1px solid #ddd' };

  return (
    <div style={{ padding: '20px', maxWidth: '1400px', margin: '0 auto' }}>
      <h1>🏢 مراكز التكلفة</h1>
      
      <button onClick={() => window.location.href = '/dashboard'} style={{ padding: '10px 20px', backgroundColor: '#6c757d', color: 'white', border: 'none', borderRadius: '4px', marginBottom: '20px', cursor: 'pointer' }}>
        رجوع للوحة التحكم
      </button>

      {message && <p style={{ padding: '10px', backgroundColor: message.includes('نجاح') ? '#d4edda' : '#f8d7da', borderRadius: '4px' }}>{message}</p>}

      <div style={{ marginBottom: '20px' }}>
        <button onClick={() => { setShowForm(!showForm); if (!showForm) fetchNextCode(); }} style={{ padding: '12px 25px', backgroundColor: showForm ? '#dc3545' : '#6f42c1', color: 'white', border: 'none', borderRadius: '8px', cursor: 'pointer', fontSize: '16px', fontWeight: 'bold' }}>
          {showForm ? '❌ إلغاء' : '➕ مركز جديد'}
        </button>
      </div>

      {showForm && (
        <form onSubmit={handleSubmit} style={{ backgroundColor: '#f8f9fa', padding: '20px', borderRadius: '8px', marginBottom: '20px', border: '2px solid #6f42c1' }}>
          <h3 style={{ color: '#6f42c1' }}>➕ مركز تكلفة جديد</h3>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: '15px' }}>
            <div>
              <label>الكود (تلقائي):</label>
              <input type="text" value={formData.center_code} readOnly style={{ width: '100%', padding: '8px', backgroundColor: '#e2e8f0' }} />
            </div>
            <div>
              <label>اسم المركز:</label>
              <input type="text" value={formData.center_name} onChange={(e) => setFormData({...formData, center_name: e.target.value})} required style={{ width: '100%', padding: '8px' }} />
            </div>
            <div>
              <label>النوع:</label>
              <select value={formData.center_type} onChange={(e) => { setFormData({...formData, center_type: e.target.value}); fetchNextCode(); }} style={{ width: '100%', padding: '8px' }}>

            ))
          )}
        </tbody>
      </table>
